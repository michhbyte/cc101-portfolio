import java.util.Scanner;
import java.util.Random;

public class GuesstheNumber {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        Random random = new Random();
        
        int secretNumber = random.nextInt(20) + 1;
        int userGuess = 0;
        int attempts = 0;
        final int MAX_ATTEMPTS = 8;
        
        System.out.println("=*=*=*=*=*=*=*=*=*=*=*=*=*=*=*=*=*=*=*=*");
        System.out.println("           Guess the Number     ");
        System.out.println("=*=*=*=*=*=*=*=*=*=*=*=*=*=*=*=*=*=*=*=*");
        System.out.println("Think a number between 1 to 20 only.");
        System.out.println("You have " + MAX_ATTEMPTS + " attempts to guess it correctly!\n");
        
        while (userGuess != secretNumber && attempts < MAX_ATTEMPTS) {
            System.out.print("Enter your guess: ");
            
            if (!scanner.hasNextInt()) {
                System.out.println(" Please enter a valid number!\n");
                scanner.next();
                continue;
            }
            
            userGuess = scanner.nextInt();
            attempts++;
            
            if (userGuess < secretNumber) {
                System.out.println("Too low! Try again.\n");
            } else if (userGuess > secretNumber) {
                System.out.println("Too high! Try again.\n");
            }
        }
        
        if (userGuess == secretNumber) {
            System.out.println("Congratulations! You finally guessed the number.");
            System.out.println(" It took you " + attempts + " attempt(s).");
        } else {
            System.out.println(" Game Over! You've used all " + MAX_ATTEMPTS + " attempts.");
            System.out.println(" The secret number was: " + secretNumber);
        }
        
        System.out.println("\nThank you for playing!");
        scanner.close();
    }
}
